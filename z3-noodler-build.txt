# BUILD MATA static
cd /tmp
rm -rf mata-musl
git clone https://github.com/VeriFIT/mata.git mata-musl
cd mata-musl

export CC=x86_64-linux-musl-gcc
export CXX=x86_64-linux-musl-g++
export AR=x86_64-linux-musl-ar
export RANLIB=x86_64-linux-musl-ranlib
export STRIP=x86_64-linux-musl-strip
export CFLAGS="-O2 -fPIC -static -U_FORTIFY_SOURCE"
export CXXFLAGS="-O2 -fPIC -static -U_FORTIFY_SOURCE"
# Optional: ensure no host pkg-config leakage
export PKG_CONFIG_PATH=""

make clean || true
make release
sudo make install

# BUILD Z3-Nooder static
set -e
set -o pipefail

cd ~/Repos/z3-noodler
rm -rf build
mkdir build
cd build

LOGFILE=build-$(date +%Y%m%d-%H%M%S).log
echo "Logging to $LOGFILE"

# Configure
cmake .. \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_C_COMPILER=x86_64-linux-musl-gcc \
  -DCMAKE_CXX_COMPILER=x86_64-linux-musl-g++ \
  -DZ3_BUILD_LIBZ3_SHARED=OFF \
  -DZ3_USE_LIB_GMP=OFF \
  -DCMAKE_C_FLAGS="-static -I/usr/local/include" \
  -DCMAKE_CXX_FLAGS="-static -I/usr/local/include" \
  -DCMAKE_EXE_LINKER_FLAGS="-static -L/usr/local/lib -Wl,--whole-archive -lmata -Wl,--no-whole-archive" \
  2>&1 | tee "$LOGFILE"
cmake_rc=${PIPESTATUS[0]}
if [ $cmake_rc -ne 0 ]; then echo "CMake failed"; exit $cmake_rc; fi

# Build
make -j"$(nproc)" 2>&1 | tee -a "$LOGFILE"
make_rc=${PIPESTATUS[0]}
exit $make_rc
